## 1. 设计目标

任务详情页是一个**独立子页面**，在 TasksPage 内部切换（不依赖 Loop 页面的左侧 Loop 列表）。

```
任务列表 → 点击任务 → 任务详情页（TaskDetailPage）
  ├── 上半：工艺要求（复用 existing data + 组件）
  └── 下半：执行历史 + 展开看板
```

## 3. 后端设计

### 3.1 API：GET /api/v1/tasks/{loop_id}

```json
{
  "code": 0,
  "data": {
    "loop": {
      "id": 12, "name": "轻量任务工艺", "description": "...",
      "status": "enabled", "workspace_id": 1
    },
    "template": {
      "name": "superpowers-task", "display_name": "轻量任务工艺",
      "complexity": "light", "version": "1.0.0"
    },
    "steps": [
      {
        "id": 29, "name": "头脑风暴", "order_index": 0,
        "skill_names": ["superpowers-brainstorm"],
        "expected_artifacts": [{"name":"brainstorm","type":"file","path":".ntd/brainstorm.md"}],
        "gate_config": [{"name":"方案存在","type":"artifact_present","artifact":"brainstorm"}]
      }
    ],
    "executions": [
      {
        "id": 3581, "status": "success", "started_at": "...", "finished_at": "...",
        "total_steps": 6, "completed_steps": 6, "failed_steps": 0
      },
      {
        "id": 3579, "status": "failed", "started_at": "...", "finished_at": "...",
        "total_steps": 6, "completed_steps": 4, "failed_steps": 2
      }
    ]
  }
}
```

实现：两个查询。

```sql
-- steps（含 skill/artifact/gate 配置）
SELECT s.id, s.name, s.order_index, s.skill_names, s.expected_artifacts, s.gate_config
FROM loop_steps s WHERE s.loop_id = ?1 ORDER BY s.order_index;

-- executions
SELECT le.id, le.status, le.started_at, le.finished_at,
       le.total_steps, le.completed_steps, le.failed_steps
FROM loop_executions le WHERE le.loop_id = ?1
ORDER BY le.started_at DESC LIMIT 20;
```

### 4.3 API 客户端

```typescript
// bundled.ts 新增
async getTaskDetail(loopId: number): Promise<{
  loop: { id, name, description, status, workspace_id };
  template: { name, display_name, complexity, version };
  steps: Array<{ id, name, order_index, skill_names, expected_artifacts, gate_config }>;
  executions: Array<{ id, status, started_at, finished_at, total_steps, completed_steps, failed_steps }>;
}> {
  return unwrap(await api.get(`/api/v1/tasks/${loopId}`));
}

// 触发新执行（触发 Loop 执行）
async triggerLoopExecution(loopId: number): Promise<{ execution_id: number }> {
  return unwrap(await api.post(`/api/v1/loops/${loopId}/trigger`, { trigger_type: "manual" }));
}
```
